returns trigger
language plpgsql
as $$
begin
  raise exception 'stacks_metadata rows cannot be deleted';
end;
$$;

create trigger tr_no_delete_stacks_metadata
before delete on public.stacks_metadata
for each row execute function public.prevent_stacks_metadata_delete();

SQL 11

-- Drop the existing blanket trigger and function
drop trigger tr_no_update_stacks_metadata on public.stacks_metadata;
drop function public.prevent_stacks_metadata_mutation() cascade;

-- New function that only blocks changes to id and created_at
create or replace function public.prevent_stacks_metadata_immutable_fields()
returns trigger
language plpgsql
as $$
begin
  if NEW.id <> OLD.id then
    raise exception 'stacks_metadata.id is immutable';
  end if;

  if NEW.created_at <> OLD.created_at then
    raise exception 'stacks_metadata.created_at is immutable';
  end if;

  return NEW;
end;
$$;

-- Re-attach trigger for updates only
create trigger tr_protect_stacks_metadata_fields
before update on public.stacks_metadata
for each row execute function public.prevent_stacks_metadata_immutable_fields();

-- Keep the delete trigger
create or replace function public.prevent_stacks_metadata_delete()
returns trigger
language plpgsql
as $$
begin
  raise exception 'stacks_metadata rows cannot be deleted';
end;
$$;

create trigger tr_no_delete_stacks_metadata
before delete on public.stacks_metadata
for each row execute function public.prevent_stacks_metadata_delete();

SQL 15

create extension if not exists pgcrypto;
create extension if not exists citext;

-- If using Supabase Auth shadow users:
-- public.users.id should equal auth.uid() (1:1 mapping)
create table if not exists public.users (
  id uuid primary key, -- equals auth.uid()
  privy_user_id text not null unique,
  wallet_address text, 
  created_at timestamptz not null default now()
);

create table if not exists public.profiles (
  user_id uuid primary key references public.users(id) on delete cascade,
  username citext unique,
  updated_at timestamptz not null default now()
);

create table if not exists public.stacks_metadata (
  id text primary key,
  stackname text,
  created_at timestamptz not null default now()
);
